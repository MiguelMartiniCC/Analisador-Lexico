int fatorial(int n) {
    int resultado, anterior;

    resultado = 1;

    if (n > 1) {
        anterior = func fatorial(n - 1);
        resultado = n * anterior;
    }

    return resultado;
}

float media(int a, int b) {
    float resultado;

    resultado = (a + b) / 2.0;

    return resultado;
}

int soma(int[3] valores) {
    return valores[0] + valores[1] + valores[2];
}

boolean maior(int a, int b) {
    boolean resultado;

    resultado = a > b;

    return resultado;
}

    int[3] valores;
    int[2][2] matriz;
